console.log("🚀 CACHE MANAGER READY (PROTECTED)");

window.CACHE_PRIORITY = {
  home: 100 // 🔥 never remove HOME first
};

window.cleanupCache = function() {

  const LIMIT = window.IS_LOW_END ? 2 : 5;

  if (!window.PAGE_CACHE) return;

  while (window.PAGE_CACHE.size > LIMIT) {

    const keys = Array.from(window.PAGE_CACHE.keys());

    let removed = false;

    for (let i = 0; i < keys.length; i++) {

      const key = keys[i];

      // 🔥 SKIP HOME ALWAYS
      if (key === "home") continue;

      window.PAGE_CACHE.delete(key);
      removed = true;
      break;
    }

    // ❌ agar sirf HOME bacha hai, break infinite loop
    if (!removed) break;
  }
};
