window.removeSkeleton = function(container) {
    if (!container) return;

    container.querySelector("#instantSkeleton")?.remove();

    container
        .querySelectorAll("[data-page-skeleton='true']")
        .forEach(el => {
            if (el.id !== "instantSkeleton") {
                el.remove();
            }
        });

    window.__INSTANT_SKELETON_ACTIVE__ = false;
};
