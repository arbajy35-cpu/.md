        position:fixed;

        left:50%;
        transform:translateX(-50%);
        background:#ff3040;
        color:#fff;

    `;

        Failed to load ${page}
        <button
            type="button"
            style="
                margin-left:8px;
                padding:4px 12px;
                background:#fff;
                color:#ff3040;
                border:0;
                border-radius:4px;
                font-weight:600
            "
        >Retry</button>
    `;

    toast.querySelector("button")?.addEventListener(
        "click",
        () => {
            toast.remove();
            window.loadPage(page, true);
        }
    );

    document.body.appendChild(toast);

    setTimeout(() => toast.remove(), 4000);
};
