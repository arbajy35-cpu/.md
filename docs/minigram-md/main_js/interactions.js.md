function commentPost(postId){

  const commentsEl =
    document.getElementById(
      "comments-" + postId
    );

  if(!commentsEl){
    return;
  }

  let count =

    parseInt(
      commentsEl.dataset.comments || "0"
    ) || 0;

  count++;

  commentsEl.dataset.comments =
    count;

  commentsEl.textContent =
    count + " comments";

  toast("Comment added");
}
