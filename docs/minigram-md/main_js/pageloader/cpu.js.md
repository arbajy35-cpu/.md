    const start = performance.now();

    setTimeout(() => {

      const delay =
        performance.now() - start;

      /*
       * Expected timer delay is roughly a few ms.
       * Higher delay = busier main thread.
       */

      const timerLoad =
        Math.max(
          0,
          Math.min(
            100,
            (delay - 4) * 5
          )
        );

            100,

      //////////////////////////////////////////////////

      const target =
        Math.max(
          timerLoad,
          taskLoad

      //////////////////////////////////////////////////
