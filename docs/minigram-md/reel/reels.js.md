    }

  });

function watchIndex(){

  let last = -1;

  function loop(){

    const s = window.REELS_STATE;
    if (!s) return requestAnimationFrame(loop);

    if (s.index !== last){
      last = s.index;
      window.render?.();
    }

    requestAnimationFrame(loop);
  }

  loop();
}
