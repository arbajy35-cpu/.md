  function drawDistantHills(offset, baseY, color, amp, freq) {

    }
    ctx.lineTo(W, H);
    ctx.closePath();
    ctx.fill();
  }

  function drawTerrain() {

    ctx.beginPath();
    for (let x = startX; x <= endX; x += TERRAIN_STEP) {
      const y = terrainHeightAt(x);
      if (x === startX) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }

  function drawWheel(x, y) {
    ctx.save();
    ctx.translate(x, y);
    
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath();
    ctx.arc(2, 2, WHEEL_RADIUS, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.fillStyle = '#1F2937';
    ctx.beginPath();
    ctx.arc(0, 0, WHEEL_RADIUS, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.fillStyle = '#E5E7EB';
    ctx.beginPath();
    ctx.arc(0, 0, WHEEL_RADIUS * 0.6, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.strokeStyle = '#6B7280';
    ctx.lineWidth = 2.5;
    ctx.save();
    ctx.rotate(car.wheelSpin);
    for (let i = 0; i < 5; i++) {
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(Math.cos(i * Math.PI * 2 / 5) * WHEEL_RADIUS * 0.55, 
                 Math.sin(i * Math.PI * 2 / 5) * WHEEL_RADIUS * 0.55);
      ctx.stroke();
    }
    ctx.restore();
    
    ctx.fillStyle = '#374151';
    ctx.beginPath();
    ctx.arc(0, 0, 4, 0, Math.PI * 2);
    ctx.fill();
    
    ctx.restore();

  function drawParticles() {
    particles.forEach(p => {
      ctx.globalAlpha = Math.max(0, p.life * 1.5);
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x - camX, p.y, p.r, 0, Math.PI * 2);
      ctx.fill();
    });
    ctx.globalAlpha = 1;
  }

  function roundRect(x, y, w, h, r, fill) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
    if (fill) ctx.fill();
  }

  function render() {
    drawBackground();
    drawTerrain();

    const minX = camX - 60;
    const maxX = camX + W + 60;
    coins.forEach(c => { if (!c.taken && c.x > minX && c.x < maxX) drawCoin(c.x, c.y); });
    fuels.forEach(f => { if (!f.taken && f.x > minX && f.x < maxX) drawFuel(f.x, f.y); });

    drawParticles();
    drawCar();
  }

  function loop(now) {
    const dt = Math.min(0.033, (now - last) / 1000 || 0);
    last = now;
    update(dt);
    render();
    requestAnimationFrame(loop);
  }

  bindInput();
  resetGame();
  render();
  requestAnimationFrame((t) => { last = t; loop(t); });
})();
