        const type = card.dataset.type;

        let filtered = [];

        if (type === "all") {
          filtered = ALL_POSTS;
        } else {
          filtered = ALL_POSTS.filter(p => p.type === type);

        //////////////////////////////////////////////////
        // 🌍 PASS DATA

          loadPage("saved_view");

        }
      };

    });

    //////////////////////////////////////////////////

    const backBtn = document.querySelector(".backBtn");
    if (backBtn) {
      backBtn.onclick = window.goBack;
    }
  }

  // 🔙 GLOBAL BACK (OUTSIDE INIT)

    console.log("🔙 Going back...");
