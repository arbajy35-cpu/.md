      return;

    // ================================
    // 📊 CALCULATE %

    bar.style.width = percent + "%";

    // 🔁 LOOP ONLY WHEN PLAYING
    // ================================
    if (!video.paused && !video.ended){
      rafId = requestAnimationFrame(update);
    }
  }

    cancelAnimationFrame(rafId);

  video.addEventListener("pause", () => {
    cancelAnimationFrame(rafId);
  });

  video.addEventListener("ended", () => {
    cancelAnimationFrame(rafId);
    bar.style.width = "100%";
  });

}
