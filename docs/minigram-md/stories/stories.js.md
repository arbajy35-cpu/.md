  story,
  index

    document.createElement("div");

    "storyItem";

  // 🎨 RING
  //////////////////////////////////////////////////

  const ring =
    document.createElement("div");

    "storyRing";

    document.createElement("img");

    "lazy";

    false;

  // 👤 USERNAME
  //////////////////////////////////////////////////

    document.createElement("div");

  user.className =
    "storyUser";

  user.textContent =
    story.user;

  //////////////////////////////////////////////////
  // 📦 BUILD
  //////////////////////////////////////////////////

  ring.appendChild(img);

  item.appendChild(ring);

  item.appendChild(user);

  //////////////////////////////////////////////////
  // 👆 CLICK EVENT
  //////////////////////////////////////////////////

  item.addEventListener(
    "click",
    () => {

      window.openStory(index);

    }
  );

  return item;
}
